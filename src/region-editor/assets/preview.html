🚀 新しいテーマの追加手順

RE（Region Editor）のテーマは src/region-editor/render/styles/themes.ts に定義されています。
基本テーマ（RegionTheme: schley, perilous, parchment, monochrome, illustrated）または、絵地図用パレット（IllustratedPalette: natural, sepia, lush, nordic）として拡張が可能です。

TypeScript 型定義 (ThemeColors)

export interface ThemeColors {
  background: string;       // 背景色
  ocean: string;            // 深海色
  oceanShallow: string;     // 沿岸浅瀬色
  coastlineStroke: string;  // 海岸線輪郭色
  riverFill: string;        // 河川塗り色
  riverStroke: string;      // 河川境界線色
  roadStroke: string;       // 街道線色
  highwayStroke: string;    // 幹線街道線色
  bridgeDeck: string;       // 橋梁路面色
  bridgeRail: string;       // 橋梁高欄色
  borderStroke: string;     // 国境線色
  textPrimary: string;      // 主地名テキスト色
  textSecondary: string;    // 補足テキスト色
  textWater: string;        // 水域名称テキスト色
  mountainFill: string;     // 山岳シャドウ・塗り色
  mountainHighlight: string;// 山岳光面・城壁塗り色
  mountainStroke: string;   // 山岳・地勢輪郭線色
  hillFill: string;         // 丘陵塗り色
  treeFill: string;         // 樹木シンボル塗り色
  treeStroke: string;       // 樹木輪郭線色
  settlementFill: string;   // 都市屋根・アクセント色
  settlementStroke: string; // 都市輪郭・窓扉色
  landmarkFill: string;     // ランドマーク色
  contourStroke: string;    // 等高線色
  contourIndexStroke: string;// 主等高線（計曲線）色
  forestCanopy: {
    deciduous: string; coniferous: string; tropical: string;
    shadow: string; highlight: string; stroke: string;
  };
  biomes: Record<BiomeKind, string>;
}
